`scripts/lib/serve.mjs` (static server used by Playwright and the a11y scan):
```js
import {createServer} from 'node:http';
import {readFile, stat} from 'node:fs/promises';
import {extname, join, normalize} from 'node:path';

const TYPES = {'.html': 'text/html', '.js': 'text/javascript', '.mjs': 'text/javascript', '.css': 'text/css',
  '.json': 'application/json', '.svg': 'image/svg+xml', '.png': 'image/png', '.woff2': 'font/woff2', '.ico': 'image/x-icon'};

export function serve(dir, port) {
  const server = createServer(async (req, res) => {
    const path = normalize(decodeURIComponent(new URL(req.url, 'http://x').pathname)).replace(/^(\.\.[/\\])+/, '');
    let file = join(dir, path);
    try { if ((await stat(file)).isDirectory()) file = join(file, 'index.html'); } catch {}
    try {
      const body = await readFile(file);
      res.writeHead(200, {'content-type': TYPES[extname(file)] ?? 'application/octet-stream'});
      res.end(body);
    } catch { res.writeHead(404); res.end('not found'); }
  });
  return new Promise(resolve => server.listen(port, '127.0.0.1', () =>
    resolve({url: `http://127.0.0.1:${port}`, close: () => server.close()})));
}

if (process.argv[1] === new URL(import.meta.url).pathname) {
  const [dir = 'dist', port = '4173'] = process.argv.slice(2);
  serve(dir, Number(port)).then(s => console.log(`serving ${dir} at ${s.url}`));
}
```

### Task 2: Tokens from Tali's Figma

**Files:** create `scripts/lib/figma.mjs`, `scripts/lib/foundations.mjs`, `scripts/lib/css.mjs`,
`scripts/figma-pull.mjs`, `scripts/build-tokens.mjs`, `tokens/figma.config.json`, `tests/tokens.test.mjs`;
generated `tokens/foundations.tokens.json`, `tokens/variable-ids.json`, `src/tokens.css`.

**Interfaces:**
- `fetchFile(fileKey) → Promise<object>` (figma.mjs; caches to `.cache/figma-<key>.json`, `--fresh` refetches).
- `parseFoundations(pageNode) → {tokens, variableIds}` where `tokens` is DTCG (groups below) and `variableIds` maps
  a Figma `VariableID:…` to a CSS custom property name (`--gv-surface-app`).
- `toCss(tokens) → string`.
- CSS names: primitives `--gv-color-<family>-<step>` (`slate/100` → `--gv-color-slate-100`, `alpha/black-70` →
  `--gv-color-alpha-black-70`); semantic `--gv-<group>-<role>` (`surface/app` → `--gv-surface-app`) as
  `var(--gv-color-…)`; `--gv-space-<n>`; `--gv-radius-<n|full>`; `--gv-font-family`; per type role
  `--gv-type-<role>` (CSS `font` shorthand) and `--gv-type-<role>-tracking`; `--gv-shadow-<name>`.
  Role slugs: `Stat small` → `stat-small`, `Body strong` → `body-strong`.

How the Foundations page is built (read on 29 Oct 2026):
- Colour: a FRAME named `<group>/<role>` holding RECTANGLE `Swatch` (its fill = the value; alpha = `fill.opacity ×
  color.a`; `boundVariables.fills[0].id` = the semantic variable) and two TEXT nodes: the role, then the primitive
  name (`slate/100`).
- Type: inside FRAME `Typography`, a FRAME per role (name = role) holding FRAME `Label` and one sample TEXT whose
  `style` gives `fontFamily`, `fontSize`, `lineHeightPx`, `fontWeight`, `letterSpacing`.
- Space: FRAME `space/<n>` holding RECTANGLE `Bar` (value = its width; id at `boundVariables.size.x.id`).
- Radius: FRAME `radius/<n|full>` holding RECTANGLE `Shape` (`cornerRadius`; 999 means full; id at
  `boundVariables.rectangleCornerRadii.RECTANGLE_TOP_LEFT_CORNER_RADIUS.id`).
- Shadow: FRAME `shadow/<name>` holding FRAME `Sample` with `effects` (DROP_SHADOW: color, offset, radius, spread).

```js
const children = n => n.children ?? [];
function* walk(n) { yield n; for (const c of children(n)) yield* walk(c); }
const slug = s => s.trim().toLowerCase().replace(/\s+/g, '-');
const px = value => ({value: Math.round(value * 100) / 100, unit: 'px'});
const hex2 = v => Math.round(v * 255).toString(16).padStart(2, '0');

function colorValue(fill) {
  const {r, g, b, a = 1} = fill.color;
  const alpha = Math.round(a * (fill.opacity ?? 1) * 1000) / 1000;
  return {colorSpace: 'srgb', components: [r, g, b].map(v => Math.round(v * 10000) / 10000), alpha,
    hex: `#${hex2(r)}${hex2(g)}${hex2(b)}`};
}
const sameColor = (x, y) => x.hex === y.hex && Math.abs(x.alpha - y.alpha) < 0.005;

export function parseFoundations(page) {
  const tokens = {color: {}, space: {}, radius: {}, type: {}, shadow: {}};
  const variableIds = {};
  for (const n of walk(page)) {
    if (n.type !== 'FRAME') continue;
    const kids = children(n);
    const rect = name => kids.find(k => k.type === 'RECTANGLE' && k.name === name);
    let m;
    if ((m = n.name.match(/^([a-z-]+)\/([a-z0-9-]+)$/)) && rect('Swatch')) {
      const [, group, role] = m;
      const prim = kids.filter(k => k.type === 'TEXT').at(-1).characters.trim();
      const [family, step] = prim.split('/');
      const value = colorValue(rect('Swatch').fills[0]);
      tokens.color[family] ??= {};
      const known = tokens.color[family][step];
      if (known && !sameColor(known.$value, value)) throw new Error(`${prim} has two values: ${known.$value.hex} and ${value.hex}`);
      tokens.color[family][step] ??= {$type: 'color', $value: value};
      tokens[group] ??= {};
      tokens[group][role] = {$type: 'color', $value: `{color.${family}.${step}}`};
      const id = rect('Swatch').boundVariables?.fills?.[0]?.id;
      if (id) variableIds[id] = `--gv-${group}-${role}`;
    } else if ((m = n.name.match(/^space\/(\d+)$/)) && rect('Bar')) {
      tokens.space[m[1]] = {$type: 'dimension', $value: px(rect('Bar').absoluteBoundingBox.width)};
      const id = rect('Bar').boundVariables?.size?.x?.id;
      if (id) variableIds[id] = `--gv-space-${m[1]}`;
    } else if ((m = n.name.match(/^radius\/(\d+|full)$/)) && rect('Shape')) {
      const r = rect('Shape').cornerRadius ?? 0;
      tokens.radius[m[1]] = {$type: 'dimension', $value: px(r >= 999 ? 9999 : r)};
      const id = rect('Shape').boundVariables?.rectangleCornerRadii?.RECTANGLE_TOP_LEFT_CORNER_RADIUS?.id;
      if (id) variableIds[id] = `--gv-radius-${m[1]}`;
    } else if ((m = n.name.match(/^shadow\/([a-z0-9-]+)$/))) {
      const sample = kids.find(k => k.name === 'Sample');
      const effects = (sample?.effects ?? []).filter(e => e.visible !== false && e.type === 'DROP_SHADOW');
      tokens.shadow[m[1]] = {$type: 'shadow', $value: effects.map(e => ({color: colorValue({color: e.color}),
        offsetX: px(e.offset.x), offsetY: px(e.offset.y), blur: px(e.radius), spread: px(e.spread ?? 0)}))};
    } else if (n.name === 'Typography') {
      for (const role of kids.filter(k => k.type === 'FRAME' && children(k).some(c => c.name === 'Label'))) {
        const s = children(role).find(c => c.type === 'TEXT').style;
        tokens.type[slug(role.name)] = {$type: 'typography', $value: {fontFamily: s.fontFamily,
          fontSize: px(s.fontSize), lineHeight: Math.round((s.lineHeightPx / s.fontSize) * 10000) / 10000,
          fontWeight: s.fontWeight, letterSpacing: px(s.letterSpacing ?? 0)}};
      }
    }
  }
  return {tokens, variableIds};
}
```

```js
const rgb = c => {
  const [r, g, b] = c.components.map(v => Math.round(v * 255));
  return c.alpha >= 1 ? c.hex : `rgb(${r} ${g} ${b} / ${c.alpha})`;

const ref = v => `var(--gv-${v.slice(1, -1).replaceAll('.', '-')})`;
const dim = d => `${d.value}${d.unit}`;

export function toCss(tokens) {
  const lines = [];
  for (const [family, steps] of Object.entries(tokens.color))
    for (const [step, t] of Object.entries(steps)) lines.push(`--gv-color-${family}-${step}: ${rgb(t.$value)};`);
  const families = new Set(Object.values(tokens.type).map(t => t.$value.fontFamily));
  if (families.size !== 1) throw new Error(`expected one font family, got ${[...families]}`);
  lines.push(`--gv-font-family: '${[...families][0]}', system-ui, sans-serif;`);
  for (const [group, roles] of Object.entries(tokens)) {
    if (['color', 'space', 'radius', 'type', 'shadow'].includes(group)) continue;
    for (const [role, t] of Object.entries(roles)) lines.push(`--gv-${group}-${role}: ${ref(t.$value)};`);
  }
  for (const [n, t] of Object.entries(tokens.space)) lines.push(`--gv-space-${n}: ${dim(t.$value)};`);
  for (const [n, t] of Object.entries(tokens.radius)) lines.push(`--gv-radius-${n}: ${dim(t.$value)};`);
  for (const [role, t] of Object.entries(tokens.type)) {
    const v = t.$value, line = Math.round(v.fontSize.value * v.lineHeight);
    lines.push(`--gv-type-${role}: ${v.fontWeight} ${dim(v.fontSize)}/${line}px var(--gv-font-family);`);
    lines.push(`--gv-type-${role}-tracking: ${dim(v.letterSpacing)};`);
  }
  for (const [name, t] of Object.entries(tokens.shadow))
    lines.push(`--gv-shadow-${name}: ${t.$value.map(s =>
      `${dim(s.offsetX)} ${dim(s.offsetY)} ${dim(s.blur)} ${dim(s.spread)} ${rgb(s.color)}`).join(', ')};`);
  return `/* Generated by scripts/build-tokens.mjs from tokens/foundations.tokens.json. Do not edit by hand. */\n` +
    `:root {\n${lines.map(l => `  ${l}`).join('\n')}\n}\n`;
}
```

```js
export async function fetchFile(fileKey = config.fileKey, {fresh = process.argv.includes('--fresh')} = {}) {
  const cache = `.cache/figma-${fileKey}.json`;
```

```js
const RANK = {minor: 1, moderate: 2, serious: 3, critical: 4};
export function summarize(raw, generatedAt = new Date().toISOString()) {
  const byTitle = new Map();
  for (const r of raw) {
    const c = byTitle.get(r.title) ?? {title: r.title, storyIds: new Set(), violations: 0, byRule: {}, worst: null, items: []};
    c.storyIds.add(r.story);
    for (const x of r.violations) {
      const n = x.nodes.length;
      c.violations += n;
      c.byRule[x.id] = (c.byRule[x.id] ?? 0) + n;
      if (!c.worst || RANK[x.impact] > RANK[c.worst]) c.worst = x.impact;
      c.items.push({story: r.story, direction: r.direction, rule: x.id, impact: x.impact, help: x.help, helpUrl: x.helpUrl, nodes: n});
    }
    byTitle.set(r.title, c);
  }
  const components = [...byTitle.values()]
    .map(({storyIds, ...c}) => ({...c, stories: storyIds.size}))
    .sort((a, b) => b.violations - a.violations || a.title.localeCompare(b.title));
  return {generatedAt, palette: "Go Vocal's own palette (no city colours yet)", components};
}
```
