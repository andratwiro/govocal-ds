# Icons

Lucide icons. Tali's Figma names them; the full set is at [lucide.dev](https://lucide.dev).

export const figmaNames = iconsJson.figma;
export const hasLucide = name => Boolean(lucide[toPascalCase(name)]);
export const inFigma = figmaNames.filter(hasLucide);
export const noMatch = figmaNames.filter(name => !hasLucide(name));
export const prototypeOnly = ['map-pin', 'map', 'layout-grid', 'eye-off', 'grip-vertical', 'calendar', 'clock',
  'globe', 'flag', 'lock', 'message-circle', 'megaphone', 'thumbs-up', 'trash-2', 'pencil', 'external-link',
  'panel-left-close', 'panel-left-open'];
export const IconPair = ({name}) => {
  const Icon = lucide[toPascalCase(name)];
  return <>
    <Icon size={16} aria-hidden="true" />
    <span style={{marginInlineStart: 8}}><Icon size={20} aria-hidden="true" /></span>
  </>;
};
